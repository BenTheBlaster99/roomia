pythonfrom __future__ import annotations
import torch
import numpy as np
from PIL import Image


class SegmentationModel:
    def __init__(self):
        print("Loading SAM2...")
        try:
            from sam2.build_sam import build_sam2
            from sam2.sam2_image_predictor import SAM2ImagePredictor

            checkpoint = "checkpoints/sam2_hiera_large.pt"
            cfg = "sam2_hiera_l.yaml"
            sam = build_sam2(cfg, checkpoint, device="cuda")
            self.predictor = SAM2ImagePredictor(sam)
            print("SAM2 ready on CUDA")
        except Exception as e:
            print(f"SAM2 failed to load: {e}")
            self.predictor = None

    def segment(
        self,
        image: Image.Image,
        norm_x: float,   # 0-1 fraction of width
        norm_y: float,   # 0-1 fraction of height
    ) -> tuple[Image.Image, Image.Image]:
        """Returns (mask_L, overlay_RGB)"""
        if self.predictor is None:
            raise RuntimeError("SAM2 not loaded")

        img_rgb = np.array(image.convert("RGB"))
        h, w = img_rgb.shape[:2]
        px, py = int(norm_x * w), int(norm_y * h)

        with torch.inference_mode(), torch.autocast("cuda", dtype=torch.bfloat16):
            self.predictor.set_image(img_rgb)
            masks, scores, _ = self.predictor.predict(
                point_coords=np.array([[px, py]]),
                point_labels=np.array([1]),
                multimask_output=True,
            )

        best = masks[np.argmax(scores)]

        # Grayscale mask
        mask_img = Image.fromarray((best * 255).astype(np.uint8), mode="L")

        # RGBA overlay — red highlight on selection
        overlay = image.convert("RGBA").copy()
        highlight = np.zeros((*img_rgb.shape[:2], 4), dtype=np.uint8)
        highlight[best] = [220, 60, 60, 150]
        overlay.alpha_composite(Image.fromarray(highlight, "RGBA"))

        return mask_img, overlay.convert("RGB")

pythonfrom __future__ import annotations
import torch
import numpy as np
from PIL import Image
import scipy.ndimage as ndimage


class InpaintingModel:
    def __init__(self):
        print("Loading SD Inpainting pipeline...")
        try:
            from diffusers import StableDiffusionInpaintPipeline
            self.pipe = StableDiffusionInpaintPipeline.from_pretrained(
                "runwayml/stable-diffusion-inpainting",
                torch_dtype=torch.float16,
                safety_checker=None,
            ).to("cuda")
            self.pipe.enable_attention_slicing()
            print("SD Inpainting ready")
        except Exception as e:
            print(f"SD Inpainting failed: {e}")
            self.pipe = None

    def _prep(
        self,
        image: Image.Image,
        mask: Image.Image,
        dilate: int = 0,
    ) -> tuple[Image.Image, Image.Image, tuple[int, int]]:
        orig = image.size
        img512 = image.resize((512, 512), Image.LANCZOS)
        m = np.array(mask.resize((512, 512), Image.NEAREST))
        if dilate:
            m = ndimage.binary_dilation(m > 127, iterations=dilate).astype(np.uint8) * 255
        else:
            m = (m > 127).astype(np.uint8) * 255
        return img512, Image.fromarray(m, "L"), orig

    def remove(self, image: Image.Image, mask: Image.Image) -> Image.Image:
        if self.pipe is None:
            raise RuntimeError("Inpainting not loaded")
        img, msk, orig = self._prep(image, mask, dilate=8)
        result = self.pipe(
            prompt="empty room interior, clean background, no objects",
            negative_prompt="furniture, sofa, chair, table, clutter, objects",
            image=img,
            mask_image=msk,
            num_inference_steps=30,
            guidance_scale=7.5,
        ).images[0]
        return result.resize(orig, Image.LANCZOS)

    def replace(
        self,
        image: Image.Image,
        mask: Image.Image,
        prompt: str,
        negative_prompt: str = "ugly, blurry, distorted, low quality, deformed",
    ) -> Image.Image:
        if self.pipe is None:
            raise RuntimeError("Inpainting not loaded")
        img, msk, orig = self._prep(image, mask, dilate=4)
        full_prompt = (
            f"{prompt}, interior design, professional photography, "
            "realistic, consistent lighting, high quality"
        )
        result = self.pipe(
            prompt=full_prompt,
            negative_prompt=negative_prompt,
            image=img,
            mask_image=msk,
            num_inference_steps=35,
            guidance_scale=8.0,
        ).images[0]
        return result.resize(orig, Image.LANCZOS)

pythonfrom __future__ import annotations
import torch
import numpy as np
from PIL import Image
from typing import Tuple, Dict


class DepthModel:
    def __init__(self):
        print("Loading Depth Anything V2...")
        try:
            from transformers import pipeline
            self.pipe = pipeline(
                task="depth-estimation",
                model="depth-anything/Depth-Anything-V2-Small-hf",
                device=0 if torch.cuda.is_available() else -1,
            )
            print("Depth Anything V2 ready")
        except Exception as e:
            print(f"Depth model failed: {e}")
            self.pipe = None

    def _estimate_dims(self, depth: np.ndarray, img_w: int, img_h: int) -> Dict[str, float]:
        h, w = depth.shape

        # Normalize to 0-1 (higher = farther for Depth Anything)
        d_min, d_max = depth.min(), depth.max()
        norm = (depth - d_min) / (d_max - d_min + 1e-6)

        # Map to real-world range 0.3m–8m
        depth_m = 0.3 + norm * 7.7

        # Room length = far-wall depth, center region, 80th percentile
        cx0, cx1 = w // 4, 3 * w // 4
        cy0, cy1 = h // 4, 3 * h // 4
        room_length = float(np.percentile(depth_m[cy0:cy1, cx0:cx1], 80))

        # Width from 70° horizontal FOV (typical phone)
        fov_h = np.radians(70)
        room_width = 2 * room_length * np.tan(fov_h / 2)

        # Height from vertical FOV
        aspect = img_w / img_h
        fov_v = 2 * np.arctan(np.tan(fov_h / 2) / aspect)
        room_height = 2 * room_length * np.tan(fov_v / 2)

        return {
            "width_m":  round(float(np.clip(room_width,  2.0, 12.0)), 1),
            "length_m": round(float(np.clip(room_length, 2.0, 12.0)), 1),
            "height_m": round(float(np.clip(room_height, 2.4,  4.0)), 1),
        }

    def _colorize(self, depth: np.ndarray) -> Image.Image:
        import matplotlib.cm as cm
        norm = (depth - depth.min()) / (depth.max() - depth.min() + 1e-6)
        colored = (cm.plasma(norm)[:, :, :3] * 255).astype(np.uint8)
        return Image.fromarray(colored)

    def estimate(self, image: Image.Image) -> Tuple[Image.Image, Dict[str, float]]:
        if self.pipe is None:
            raise RuntimeError("Depth model not loaded")

        orig_w, orig_h = image.size

        # Downscale large images for speed
        max_dim = 768
        if max(orig_w, orig_h) > max_dim:
            scale = max_dim / max(orig_w, orig_h)
            image = image.resize((int(orig_w * scale), int(orig_h * scale)), Image.LANCZOS)

        result = self.pipe(image)
        depth_arr = np.array(result["depth"], dtype=np.float32)

        dims = self._estimate_dims(depth_arr, orig_w, orig_h)
        visual = self._colorize(depth_arr)

        return visual, dims

# ── Helpers ───────────────────────────────────────────────────────────────────

def b64_to_pil(b64: str) -> Image.Image:
    return Image.open(io.BytesIO(base64.b64decode(b64))).convert("RGB")

def pil_to_b64(img: Image.Image, fmt: str = "PNG") -> str:
    buf = io.BytesIO()
    img.save(buf, format=fmt)
    return base64.b64encode(buf.getvalue()).decode()

# ── Request models ────────────────────────────────────────────────────────────

class SegReq(BaseModel):
    image_base64: str
    x: float   # 0-1
    y: float   # 0-1

class RemoveReq(BaseModel):
    image_base64: str
    mask_base64: str

class ReplaceReq(BaseModel):
    image_base64: str
    mask_base64: str
    prompt: str
    negative_prompt: str = "ugly, blurry, distorted, low quality, deformed"

class DepthReq(BaseModel):
    image_base64: str

# ── Routes ────────────────────────────────────────────────────────────────────

@app.get("/health")
def health():
    return {"status": "ok"}

@app.post("/segment")
async def segment(req: SegReq):
    try:
        image = b64_to_pil(req.image_base64)
        mask, overlay = seg().segment(image, req.x, req.y)
        return {
            "mask_base64":    pil_to_b64(mask),
            "overlay_base64": pil_to_b64(overlay, "JPEG"),
        }
    except Exception as e:
        raise HTTPException(500, str(e))

@app.post("/inpaint/remove")
async def remove(req: RemoveReq):
    try:
        image = b64_to_pil(req.image_base64)
        mask  = Image.open(io.BytesIO(base64.b64decode(req.mask_base64))).convert("L")

        return {"result_base64": pil_to_b64(result, "JPEG")}
    except Exception as e:
        raise HTTPException(500, str(e))

@app.post("/inpaint/replace")

type Stage =
  | 'idle'
  | 'uploaded'
  | 'segmenting'
  | 'segmented'
  | 'selecting'
  | 'processing'
  | 'done'

function imgToB64(file: File): Promise<string> {
  return new Promise((res, rej) => {
    const r = new FileReader()
    r.onload = () => res((r.result as string).split(',')[1])
    r.onerror = rej
    r.readAsDataURL(file)
  })
}

export default function PhotoStudioPage() {
  const [stage, setStage] = useState<Stage>('idle')
  const [originalB64, setOriginalB64]     = useState('')
  const [originalSrc, setOriginalSrc]     = useState('')
  const [overlaySrc, setOverlaySrc]       = useState('')
  const [maskB64, setMaskB64]             = useState('')
  const [resultSrc, setResultSrc]         = useState('')
  const [error, setError]                 = useState<string | null>(null)
  const imgRef = useRef<HTMLImageElement>(null)
  const fileRef = useRef<HTMLInputElement>(null)

  // ── Upload ──────────────────────────────────────────────────────────────────
  async function handleUpload(e: React.ChangeEvent<HTMLInputElement>) {
    const file = e.target.files?.[0]
    if (!file) return
    const b64 = await imgToB64(file)
    setOriginalB64(b64)
    setOriginalSrc(`data:image/jpeg;base64,${b64}`)
    setStage('uploaded')
    setError(null)
  }

  // ── Click to select furniture ───────────────────────────────────────────────
  async function handleClick(e: React.MouseEvent<HTMLImageElement>) {
    if (stage !== 'uploaded') return
    const rect = e.currentTarget.getBoundingClientRect()
    const x = (e.clientX - rect.left) / rect.width
    const y = (e.clientY - rect.top)  / rect.height
    setStage('segmenting')
    setError(null)

    try {
      const res = await fetch(`${AI_URL}/segment`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ image_base64: originalB64, x, y }),
      })
      const data = await res.json()
      if (!res.ok) throw new Error(data.detail ?? 'Segmentation failed')
      setMaskB64(data.mask_base64)
      setOverlaySrc(`data:image/jpeg;base64,${data.overlay_base64}`)
      setStage('segmented')
    } catch (err: any) {
      setError(err.message)
      setStage('uploaded')
    }
  }

  // ── Remove ──────────────────────────────────────────────────────────────────
  async function handleRemove() {
    setStage('processing')
    setError(null)
    try {
      const res = await fetch(`${AI_URL}/inpaint/remove`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ image_base64: originalB64, mask_base64: maskB64 }),
      })
      const data = await res.json()
      if (!res.ok) throw new Error(data.detail ?? 'Removal failed')
      setResultSrc(`data:image/jpeg;base64,${data.result_base64}`)
      setStage('done')
    } catch (err: any) {
      setError(err.message)
      setStage('segmented')
    }
  }

  // ── Replace ─────────────────────────────────────────────────────────────────
  async function handleReplace(prompt: string) {
    setStage('processing')
    setError(null)
    try {
      const res = await fetch(`${AI_URL}/inpaint/replace`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          image_base64: originalB64,
          mask_base64: maskB64,
          prompt,
        }),
      })
      const data = await res.json()
      if (!res.ok) throw new Error(data.detail ?? 'Replacement failed')
      setResultSrc(`data:image/jpeg;base64,${data.result_base64}`)
      setStage('done')
    } catch (err: any) {
      setError(err.message)
      setStage('segmented')
    }
  }

  function reset() {
    setStage('idle')
    setOriginalB64('')
    setOriginalSrc('')
    setOverlaySrc('')
    setMaskB64('')
    setResultSrc('')
    setError(null)
    if (fileRef.current) fileRef.current.value = ''
  }

  return (
    <div className="min-h-screen bg-zinc-950 text-white">
      <nav className="flex items-center justify-between px-6 py-4 border-b border-zinc-800">
        <Link href="/" className="text-xl font-bold text-amber-400">roomia</Link>
        <span className="text-sm text-zinc-400">AI Photo Studio</span>
      </nav>

        <div>
          <h1 className="text-2xl font-bold mb-1">Edit your room photo</h1>
          <p className="text-sm text-zinc-400">
            Upload a photo of your room, click on any furniture to select it,
            then remove it or replace it with something from our catalog.
          </p>
        </div>

        {error && (
          <div className="bg-red-900/30 border border-red-700 rounded-xl px-4 py-3 text-sm text-red-300">
            {error}
          </div>
        )}

        {/* ── IDLE ── */}
        {stage === 'idle' && (
          <div
            className="border-2 border-dashed border-zinc-700 rounded-2xl p-12
                       flex flex-col items-center gap-4 hover:border-amber-400
                       transition-colors cursor-pointer"
            onClick={() => fileRef.current?.click()}
          >

app/room-capture/page.tsx
tsx'use client'

import { useRef, useState } from 'react'
import { useRouter } from 'next/navigation'
import Link from 'next/link'

const AI_URL = process.env.NEXT_PUBLIC_AI_BACKEND_URL ?? 'http://localhost:8000'

interface RoomDims {
  width_m: number
  length_m: number
  height_m: number
}

type Stage = 'idle' | 'analyzing' | 'done'

function imgToB64(file: File): Promise<string> {
  return new Promise((res, rej) => {
    const r = new FileReader()
    r.onload = () => res((r.result as string).split(',')[1])
    r.onerror = rej
    r.readAsDataURL(file)
  })
}

export default function RoomCapturePage() {
  const router = useRouter()
  const fileRef = useRef<HTMLInputElement>(null)
  const [stage, setStage] = useState<Stage>('idle')
  const [previews, setPreviews] = useState<string[]>([])
  const [depthSrc, setDepthSrc] = useState('')
  const [dims, setDims] = useState<RoomDims>({ width_m: 5, length_m: 6, height_m: 2.8 })
  const [error, setError] = useState<string | null>(null)

  async function handleUpload(e: React.ChangeEvent<HTMLInputElement>) {
    const files = Array.from(e.target.files ?? [])
    if (!files.length) return
    setError(null)
    setStage('analyzing')

    const previewUrls = files.map(f => URL.createObjectURL(f))
    setPreviews(previewUrls)

    try {
      // Use the first photo for depth estimation
      const b64 = await imgToB64(files[0])
      const res = await fetch(`${AI_URL}/depth`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ image_base64: b64 }),
      })
      const data = await res.json()
      if (!res.ok) throw new Error(data.detail ?? 'Depth estimation failed')

      setDepthSrc(`data:image/jpeg;base64,${data.depth_base64}`)
      setDims(data.room_dims)
      setStage('done')
    } catch (err: any) {
      setError(err.message)
      setStage('idle')
    }
  }

  function openInStudio() {
    router.push(
      `/studio?width=${dims.width_m}&length=${dims.length_m}&height=${dims.height_m}`
    )
  }

  return (
    <div className="min-h-screen bg-zinc-950 text-white">
      <nav className="flex items-center justify-between px-6 py-4 border-b border-zinc-800">
        <Link href="/" className="text-xl font-bold text-amber-400">roomia</Link>
        <span className="text-sm text-zinc-400">Scan Your Room</span>
      </nav>

      <div className="max-w-2xl mx-auto px-4 py-10 space-y-6">
        <div>
          <h1 className="text-2xl font-bold mb-1">Scan your room</h1>
          <p className="text-sm text-zinc-400">
            Upload 1-3 photos of your room. AI estimates the dimensions and opens
            your exact space in the 3D studio — ready to furnish.
          </p>
        </div>

        {error && (
          <div className="bg-red-900/30 border border-red-700 rounded-xl px-4 py-3
                          text-sm text-red-300">
            {error}
          </div>
        )}

        {/* ── IDLE ── */}
        {stage === 'idle' && (
          <div
            className="border-2 border-dashed border-zinc-700 rounded-2xl p-12
                       flex flex-col items-center gap-4 hover:border-amber-400
                       transition-colors cursor-pointer"
            onClick={() => fileRef.current?.click()}
          >
            <div className="text-4xl">📷</div>
            <div className="text-sm font-medium text-zinc-300">Upload room photos</div>
            <div className="text-xs text-zinc-600 text-center max-w-xs">
              Take 1-3 photos from different angles. Include the full room —
              floor to ceiling. No mirror shots.
            </div>
            <input ref={fileRef} type="file" accept="image/*" multiple
                   onChange={handleUpload} className="hidden" />
          </div>
        )}

        {/* ── ANALYZING ── */}
        {stage === 'analyzing' && (
          <div className="space-y-4">
            {previews.length > 0 && (
              <div className={`grid gap-3 ${previews.length > 1 ? 'grid-cols-2' : 'grid-cols-1'}`}>
                {previews.map((src, i) => (
                  <img key={i} src={src} alt={`Room photo ${i + 1}`}
                       className="w-full rounded-xl border border-zinc-800 opacity-60" />
                ))}
              </div>
            )}
            <div className="flex items-center justify-center gap-3 py-4">
              <span className="w-6 h-6 border-2 border-amber-400 border-t-transparent
                               rounded-full animate-spin" />
              <span className="text-sm text-amber-400">Estimating room dimensions...</span>
            </div>
          </div>
        )}

        {/* ── DONE ── */}
        {stage === 'done' && (
          <div className="space-y-6">
            {/* Photos + depth */}
            <div className="grid grid-cols-2 gap-3">
              <div>
                <p className="text-xs text-zinc-500 mb-1.5">Your photo</p>
                {previews[0] && (
                  <img src={previews[0]} alt="Room"
                       className="w-full rounded-xl border border-zinc-800" />
                )}
              </div>
              <div>
                <p className="text-xs text-zinc-500 mb-1.5">Depth map</p>
                {depthSrc && (
                  <img src={depthSrc} alt="Depth"
                       className="w-full rounded-xl border border-zinc-800" />
                )}
              </div>
            </div>

            {/* Estimated dimensions — editable */}
            <div className="bg-zinc-900 border border-zinc-800 rounded-2xl p-5 space-y-4">
              <div>
                <p className="text-sm font-semibold mb-0.5">Estimated Room Dimensions</p>
                <p className="text-xs text-zinc-500">
                  AI estimates are approximate. Adjust if needed.
                </p>
              </div>

              <div className="grid grid-cols-3 gap-4">
                {[
                  { label: 'Width', key: 'width_m', unit: 'm' },
                  { label: 'Length', key: 'length_m', unit: 'm' },
                  { label: 'Height', key: 'height_m', unit: 'm' },
                ].map(({ label, key, unit }) => (
                  <div key={key}>
                    <label className="text-xs text-zinc-500 block mb-1">{label}</label>
                    <div className="flex items-center gap-1.5">
                      <input
                        type="number"
                        min="1"
                        max="20"
                        step="0.1"
                        value={dims[key as keyof RoomDims]}
                        onChange={e => setDims(d => ({
                          ...d,
                          [key]: parseFloat(e.target.value) || 1,
                        }))}
                        className="flex-1 bg-zinc-800 border border-zinc-700 rounded-lg
                                   px-3 py-2 text-sm text-white focus:outline-none
                                   focus:border-amber-400 transition-colors"
                      />
                      <span className="text-xs text-zinc-500">{unit}</span>
                    </div>
                  </div>
                ))}
              </div>

              <p className="text-xs text-zinc-600">
                Floor area: {(dims.width_m * dims.length_m).toFixed(1)} m²
              </p>
            </div>

            {/* Actions */}
            <button
              onClick={openInStudio}
              className="w-full py-3.5 bg-amber-400 text-zinc-950 rounded-xl text-sm
                         font-bold hover:bg-amber-300 transition-colors"
            >
              🧊 Open My Room in 3D Studio →
            </button>

            <button
              onClick={() => { setStage('idle'); setPreviews([]); setDepthSrc('') }}
              className="w-full py-2 text-xs text-zinc-600 hover:text-zinc-400 transition-colors"
            >
              Scan different room
            </button>
          </div>
        )}
      </div>
    </div>
  )
}

How to Run Everything
Terminal 1 — Python backend:
bashcd roomia-ai-backend
source venv/bin/activate
uvicorn main:app --host 0.0.0.0 --port 8000
Terminal 2 — Roomia:
bashcd roomia
npm run dev
Then test in this order:

localhost:8000/health — confirms Python server is up
/room-capture — upload a room photo, check the depth map and dimensions
Click "Open in Studio" — confirms the dimension handoff works
/photo-studio — upload a room photo, click a piece of furniture, check the mask
Try remove, then try replace

The first load of each model (SAM2, SD, Depth) downloads weights and takes 2-5 minutes. After that they're cached and fast.
